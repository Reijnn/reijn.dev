{{- /* Returns the original cover resource, or nil. Cover: front.* in the page bundle first (not cover.*: Hugo's opengraph template would add that as og:image before the share card), then Open Library by ISBN. Fetched at build time, so pages make no third-party request and the CSP needs no change. Call with partialCached, keyed by the page. */ -}} {{- $img := .Resources.GetMatch "front.*" -}} {{- $title := .Title -}} {{- with and (not $img) .Params.isbn -}} {{- /* default=false makes Open Library answer 404 instead of a blank 1×1 image */ -}} {{- $url := printf "https://covers.openlibrary.org/b/isbn/%s-L.jpg?default=false" (replaceRE `[^0-9Xx]` "" (string .)) -}} {{- with try (resources.GetRemote $url) -}} {{- with .Err -}} {{- warnf "No cover for %s (%s): %s" $title $url . -}} {{- else with .Value -}} {{- $img = . -}} {{- else -}} {{- warnf "No cover for %s: Open Library has none for this ISBN, add a front.jpg to the bundle" $title -}} {{- end -}} {{- end -}} {{- end -}} {{- return $img -}}
